        return 'Hello ' + milkman + ', today (' + formattedDate + ') is Naga (No Milk) for ' + address + '. Please note this in my account. Thank you.';
      }
      return 'Hello ' + milkman + ', milk received today (' + formattedDate + '): ' + qty + ' ' + unit + ' by ' + address + ' \nPlease note this in my account. Thank you.';
    }

    function showToast(msg) {
      const t = document.getElementById('toast');
      t.textContent = msg;
      t.classList.add('show');
      setTimeout(function() { t.classList.remove('show'); }, 2500);
    }

    function setSyncStatus(statusText) {
      syncBadgeEl.textContent = statusText;
    }

    function updateLivePreviews() {
      const qty = parseFloat(entryQtyEl.value) || 0;
      const rate = parseFloat(settings.rate) || 0;
      liveCostEl.textContent = '₹' + (qty * rate).toFixed(2);

      const dateVal = entryDateEl.value || (yyyy + '-' + mm + '-' + dd);
      liveMsgPreviewEl.textContent = buildWhatsAppMessage(dateVal, qty, currentUnit, false);
    }

    function refreshUI() {
      const cleanPhone = (settings.phone || '').replace(/[^0-9]/g, '');
      const hasDrive = settings.scriptUrl && settings.scriptUrl.startsWith('https://script.google.com');
      setSyncStatus(hasDrive ? '☁️ Drive Connected' : '📱 Local Storage');
      document.getElementById('setupWarning').style.display = (cleanPhone.length >= 10 && hasDrive) ? 'none' : 'block';

      document.getElementById('driveScriptUrl').value = settings.scriptUrl || '';
      document.getElementById('milkmanName').value = settings.name || '';
      document.getElementById('milkmanPhone').value = settings.phone || '+91';
      document.getElementById('customerAddress').value = settings.address || DEFAULT_ADDRESS;
      document.getElementById('milkRate').value = settings.rate || '';

      updateLivePreviews();
      renderSummary();
      renderHistory();
    }

    async function pushToGoogleDrive(payload) {
      if (!settings.scriptUrl || !settings.scriptUrl.startsWith('https://')) return false;
      setSyncStatus('⏳ Syncing...');
      try {
        await fetch(settings.scriptUrl, {
          method: 'POST',
          mode: 'no-cors',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
        setSyncStatus('☁️ Saved to Drive');
        return true;
      } catch (err) {
        console.error(err);
        setSyncStatus('⚠️ Offline Saved');
        return false;
      }
    }

    async function pullFromGoogleDrive() {
      if (!settings.scriptUrl || !settings.scriptUrl.startsWith('https://')) {
        showToast('Add Google Script URL in Settings first');
        return;
      }
      setSyncStatus('⏳ Fetching...');
      try {
        const res = await fetch(settings.scriptUrl + '?action=get');
        const json = await res.json();
        if (json.status === 'success' && Array.isArray(json.entries)) {
          entries = json.entries.sort(function(a, b) { return b.date.localeCompare(a.date); });
          localStorage.setItem('milk_entries', JSON.stringify(entries));
          refreshUI();
          showToast('Synced with Google Drive!');
        }
      } catch (err) {
        console.error(err);
        setSyncStatus('⚠️ Sync Failed');
        showToast('Could not fetch from Drive');
      }
    }

    function sendWhatsAppMessage(text) {
      const cleanPhone = (settings.phone || '').replace(/[^0-9]/g, '');
      if (cleanPhone.length < 10) {
        alert("Please set your milkman's valid WhatsApp number in Settings first.");
        switchTab('tab-settings');
        return;
      }
      const url = 'https://wa.me/' + cleanPhone + '?text=' + encodeURIComponent(text);
      window.open(url, '_blank');
    }

    async function saveEntry(isNaga, sendWa) {
      const dateVal = entryDateEl.value;
      const qtyVal = isNaga ? 0 : (parseFloat(entryQtyEl.value) || 0);
      if (!dateVal) {
        alert('Please enter a valid date.');
        return;
      }

      const rate = parseFloat(settings.rate) || 0;
      const newEntry = {
        id: Date.now(),
        date: dateVal,
        qty: qtyVal,
        unit: currentUnit,
        rate: rate,
        amount: qtyVal * rate,
        isNaga: isNaga
      };

      const existingIdx = entries.findIndex(function(e) { return e.date === dateVal; });
      if (existingIdx > -1) {
        newEntry.id = entries[existingIdx].id;
        entries[existingIdx] = newEntry;
      } else {
        entries.push(newEntry);
      }
      entries.sort(function(a, b) { return b.date.localeCompare(a.date); });
      localStorage.setItem('milk_entries', JSON.stringify(entries));
      refreshUI();

      const synced = await pushToGoogleDrive({ action: 'save', entry: newEntry });
      showToast(synced ? 'Saved to Google Drive!' : 'Saved locally!');

      if (sendWa) {
        const msg = buildWhatsAppMessage(dateVal, qtyVal, currentUnit, isNaga);
        sendWhatsAppMessage(msg);
      }
    }

    function renderSummary() {
      const selectedMonth = summaryMonthEl.value;
      const filtered = entries.filter(function(e) { return e.date.startsWith(selectedMonth); });
      const activeEntries = filtered.filter(function(e) { return !e.isNaga; });
      const totalQty = activeEntries.reduce(function(sum, e) { return sum + e.qty; }, 0);
      const totalAmt = activeEntries.reduce(function(sum, e) { return sum + e.amount; }, 0);

      document.getElementById('statQty').textContent = totalQty + ' ' + currentUnit;
      document.getElementById('statAmount').textContent = '₹' + totalAmt.toFixed(0);
      document.getElementById('statDays').textContent = activeEntries.length + ' days (' + filtered.length + ' total)';
    }

    function renderHistory() {
      const list = document.getElementById('historyList');
      if (entries.length === 0) {
        list.innerHTML = '<p style="text-align:center; color:var(--text-muted); padding:20px 0; font-size:0.9rem;">No milk entries logged yet.</p>';
        return;
      }

      list.innerHTML = entries.map(function(e) {
        const subText = e.isNaga ? '<span style="color:#D97706; font-weight:700;">⚠️ Naga (No Milk)</span>' : (e.qty + ' ' + e.unit + ' • ₹' + Number(e.amount).toFixed(2));
        return '<div class="entry-item">' +
          '<div>' +
            '<div class="entry-date">' + formatFormattedDate(e.date) + '</div>' +
            '<div class="entry-sub">' + subText + '</div>' +
          '</div>' +
          '<div class="entry-actions">' +
            '<button type="button" class="icon-btn wa" title="Resend WhatsApp" onclick="resendEntry(' + e.id + ')">📲</button>' +
            '<button type="button" class="icon-btn del" title="Delete" onclick="deleteEntry(' + e.id + ')">🗑️</button>' +
          '</div>' +
        '</div>';
      }).join('');
    }

    window.resendEntry = function(id) {
      const e = entries.find(function(item) { return item.id === id; });
      if (!e) return;
      const msg = buildWhatsAppMessage(e.date, e.qty, e.unit, e.isNaga);
      sendWhatsAppMessage(msg);
    };

    window.deleteEntry = async function(id) {
      if (!confirm('Delete this milk entry from app and Google Drive?')) return;
      entries = entries.filter(function(item) { return item.id !== id; });
      localStorage.setItem('milk_entries', JSON.stringify(entries));
      refreshUI();
      await pushToGoogleDrive({ action: 'delete', id: id });
      showToast('Entry deleted');
    };
